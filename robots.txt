# app.ocufuse.com: no page of the web app is indexed; ocufuse.com owns search. Every page says noindex (its
# <meta name="robots"> and the X-Robots-Tag header), so crawling stays allowed: a crawler has to read a page to see
# its noindex. Everything after sign-in is private and lives on the same page (#routes). Password reset links are
# private, one-time addresses.
User-agent: *
Allow: /
Disallow: /reset-password
